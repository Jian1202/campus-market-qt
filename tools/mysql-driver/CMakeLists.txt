cmake_minimum_required(VERSION 3.21)
project(CampusMySqlDriver LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11.1 EXACT REQUIRED COMPONENTS Core Sql CorePrivate SqlPrivate)
set(MYSQL_ROOT "C:/Program Files/MySQL/MySQL Server 8.4" CACHE PATH "MySQL 安装目录")
find_path(MYSQL_INCLUDE_DIR mysql.h HINTS "${MYSQL_ROOT}/include" REQUIRED)
find_library(MYSQL_LIBRARY NAMES libmysql mysqlclient HINTS "${MYSQL_ROOT}/lib" REQUIRED)
add_library(qsqlmysql MODULE main.cpp qsql_mysql.cpp qsql_mysql_p.h)
target_include_directories(qsqlmysql PRIVATE "${MYSQL_INCLUDE_DIR}")
target_link_libraries(qsqlmysql PRIVATE Qt6::Core Qt6::Sql Qt6::CorePrivate Qt6::SqlPrivate "${MYSQL_LIBRARY}")
target_compile_definitions(qsqlmysql PRIVATE QT_PLUGIN)
set_target_properties(qsqlmysql PROPERTIES PREFIX "")
